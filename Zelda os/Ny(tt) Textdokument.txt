    <script>
        // ANSLUT TILL DIN DOCKER-BACKEND
        const socket = io('http://localhost:8080'); // Ändra till server-IP vid behov
        const logDiv = document.getElementById('log');
        const connStatus = document.getElementById('conn-status');

        function addLog(msg) {
            const time = new Date().toLocaleTimeString();
            logDiv.innerHTML = `[${time}] ${msg}<br>` + logDiv.innerHTML;
        }

        socket.on('connect', () => {
            connStatus.innerText = "DARKSIRI: ONLINE";
            connStatus.style.borderColor = "#22c55e";
            addLog("Ansluten till Modifiering Siri Darksiri.");
        });

        socket.on('disconnect', () => {
            connStatus.innerText = "DARKSIRI: OFFLINE";
            connStatus.style.borderColor = "#ef4444";
        });

        // LYSSNA PÅ UPDATERINGAR
        socket.on('asset_update', (data) => {
            addLog(`Mottog data: ${data.asset} uppdaterad.`);
            
            if(data.asset === 'Aurum') {
                document.getElementById('aurum-val').innerText = data.price + ' SEK';
                document.getElementById('aurum-progress').style.width = data.percent + '%';
            }
            
            if(data.asset === 'Creedy') {
                document.getElementById('creedy-val').innerText = data.rate + '%';
            }

            // Trigger för "P" (Profit)
            if(data.status === 'P') {
                document.body.style.boxShadow = "inset 0 0 100px rgba(168, 85, 247, 0.4)";
                setTimeout(() => document.body.style.boxShadow = "none", 2000);
            }
        });
    </script>
